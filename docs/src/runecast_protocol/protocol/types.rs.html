74/// A single cell in the game grid.
75#[derive(Debug, Clone, Serialize, Deserialize)]
76pub struct GridCell {
77    pub letter: char,
78    pub value: u8,
79    #[serde(skip_serializing_if = "Option::is_none")]
80    pub multiplier: Option<Multiplier>,
81    #[serde(default)]
82    pub has_gem: bool,
83    /// True when this cell is a "hole" — occupies its position in the
84    /// layout but can't be selected as part of a word path. Used by
85    /// Adventure Mode levels that want asymmetric playable regions.
86    /// Hole cells have no letter contribution and are skipped by the
87    /// refill pipeline.
88    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
89    pub is_hole: bool,
90    /// True when this cell has been **poisoned** by a Snake adventure
91    /// event. Poisoned letters contribute negative value when used in
92    /// a word (sign-flipped — `value` becomes `-value` for scoring),
93    /// so players want to avoid them. Poison clears when the cell is
94    /// rerolled (word consumption or another event hitting the cell).
95    /// Adventure-only; multiplayer sessions never set this.
96    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
97    pub is_poisoned: bool,
98    /// True when this cell has been **abducted** by a UFO adventure
99    /// event. Abducted cells are temporarily unselectable (mirrors
100    /// `is_hole` for path validation) and reroll back to normal on a
101    /// later round. Adventure-only; multiplayer sessions never set
102    /// this.
103    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
104    pub is_abducted: bool,
105}

155/// Game type for game pools within a lobby.
156///
157/// Each lobby can have multiple game pools, one per game type. Players wait in pools for matchmaking.
158/// Players join a game pool to find matches for that game type.
159#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
160#[serde(rename_all = "snake_case")]
161pub enum GameType {
162    /// Free-for-all open game (default)
163    Open,
164    /// 2v2 team game
165    TwoVTwo,
166    /// Adventure/co-op mode
167    Adventure,
168    /// Configurable unranked custom ("sandbox") game.
169    Sandbox,
170    /// Daily Challenge — solo timed run on the shared daily board.
171    Daily,
172    /// RuneBox — timed simultaneous word hunt. No turns, no bonus tiles and no
173    /// tile replacement; a word scores its letter count. All players work the
174    /// same static board at once for the round's duration.
175    RuneBox,
176    /// Forward-compat fallback for game types this client doesn't know.
177    #[serde(other)]
178    Unknown,
179}
180
181/// `Unknown`, so that a payload predating a field carrying `GameType` decodes
182/// to the forward-compat variant instead of failing outright. Deliberately not
183/// `Open`: a wrong-but-plausible default is worse than an explicitly unknown
184/// one, because it reads as a real answer.
185impl Default for GameType {
186    fn default() -> Self {
187        Self::Unknown
188    }
189}
190
191impl std::fmt::Display for GameType {
192    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
193        match self {
194            GameType::Open => write!(f, "open"),
195            GameType::TwoVTwo => write!(f, "two_v_two"),
196            GameType::Adventure => write!(f, "adventure"),
197            GameType::Sandbox => write!(f, "sandbox"),
198            GameType::Daily => write!(f, "daily"),
199            GameType::RuneBox => write!(f, "rune_box"),
200            GameType::Unknown => write!(f, "unknown"),
201        }
202    }
203}

209/// What to call a player, in the context this message was sent in.
210///
211/// Show [`runecast`](Self::runecast) when it is set, otherwise
212/// [`discord`](Self::discord). A viewer who has turned on austere nicknames is
213/// shown `discord` always. [`shown`](Self::shown) applies that rule.
214///
215/// ```
216/// use runecast_protocol::PlayerName;
217///
218/// let name = PlayerName::new("Ada", Some("Rune Queen".to_string()));
219/// assert_eq!(name.shown(false), "Rune Queen");
220/// assert_eq!(name.shown(true), "Ada");
221/// ```
222#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
223pub struct PlayerName {
224    /// The name Discord itself would show here: the player's nickname in this
225    /// server (when the lobby is in one and they have one there), else their
226    /// Discord display name, else their username.
227    pub discord: String,
228    /// The nickname the player chose in RuneCast, if any.
229    #[serde(default, skip_serializing_if = "Option::is_none")]
230    pub runecast: Option<String>,
231}
232
233impl PlayerName {
234    /// A name from its two layers.
235    #[must_use]
236    pub fn new(discord: impl Into<String>, runecast: Option<String>) -> Self {
237        Self {
238            discord: discord.into(),
239            runecast,
240        }
241    }
242
243    /// The name to display, given the viewer's austere-nicknames setting.
244    #[must_use]
245    pub fn shown(&self, austere: bool) -> &str {
246        match &self.runecast {
247            Some(chosen) if !austere && !chosen.is_empty() => chosen,
248            _ => &self.discord,
249        }
250    }
251}

405/// One player's standing in a RuneBox **match**, as of the end of some round.
406///
407/// A match is several rounds; [`ScoreInfo`] describes a single round. Both are
408/// sent on `RuneBoxRoundEnded` because they answer different questions — "who
409/// won this round" and "who is winning the match".
410///
411/// Carries no word list, by construction. RuneBox found-sets are independent,
412/// so another player's words stay worth full value to you and naming them at
413/// round end would hand over the answers for a board that is still fresh in
414/// everyone's memory.
415#[serde_with::serde_as]
416#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
417pub struct RuneBoxStanding {
418    /// Serializes as a quoted STRING, like every snowflake on this wire — an
419    /// id above 2^53 silently loses precision as a JSON number in JS.
420    #[serde_as(as = "serde_with::DisplayFromStr")]
421    pub player_id: i64,
422    /// The player's raw Discord username: the unique handle. Display
423    /// [`name`](Self::name) instead when it is present.
424    pub username: String,
425    /// What to call this player. Absent from servers older than protocol
426    /// 0.20; fall back to `username`.
427    #[serde(default, skip_serializing_if = "Option::is_none")]
428    pub name: Option<PlayerName>,
429    /// Rounds this player has won outright.
430    ///
431    /// The deciding number in a two-player match, which is best-of-3 rather
432    /// than elimination — there is nobody to eliminate until it is over. In a
433    /// three-or-more-player match the field still counts round wins, but
434    /// elimination is what decides the match, so nothing reads it.
435    pub round_wins: u8,
436    /// This player's scores summed across every round played so far.
437    pub total_score: i32,
438    /// True once this player has been eliminated and moved to spectating.
439    pub eliminated: bool,
440}

581/// State of the auto-rematch countdown after a game ends.
582///
583/// When a game ends, players are automatically re-queued and a countdown begins.
584/// Any player can trigger an early start, or opt out to return to the lobby.
585#[serde_with::serde_as]
586#[derive(Debug, Clone, Serialize, Deserialize, Default)]
587#[serde(tag = "status", rename_all = "snake_case")]
588pub enum RematchCountdownState {
589    /// No rematch countdown active
590    #[default]
591    Idle,
592
593    /// Countdown is active - new game will start when it reaches 0
594    Active {
595        /// When the countdown expires (absolute time for clock sync)
596        expires_at: chrono::DateTime<chrono::Utc>,
597        /// Seconds remaining (for initial state display)
598        seconds_remaining: u32,
599        /// Player IDs still in the rematch pool
600        #[serde_as(as = "Vec<serde_with::DisplayFromStr>")]
601        player_ids: Vec<i64>,
602        /// The game pool/game type for the rematch
603        game_type: GameType,
604    },
605
606    /// Auto-rematch is off for the lobby, or was switched off mid-countdown: no
607    /// timer. Each eligible player who presses Play Again (`TriggerRematch`)
608    /// joins `ready`; the rematch starts for `ready` only, once it reaches
609    /// `needed`. Exists only while at least `needed` players are eligible.
610    OptIn {
611        /// Players who pressed Play Again.
612        #[serde_as(as = "Vec<serde_with::DisplayFromStr>")]
613        ready: Vec<i64>,
614        /// The previous game's players who may still opt in (a superset of `ready`).
615        #[serde_as(as = "Vec<serde_with::DisplayFromStr>")]
616        eligible: Vec<i64>,
617        /// How many must be ready before the rematch starts (`REMATCH_MIN_PLAYERS`).
